Title "PROGRAM TO CONVERT WORD FROM UPPER CASE LETTER TO LOWER CASE LETTER"
.model small
.stack 100
.DATA
      LF EQU 10
      CR EQU 13
       
MSG1 DB "WELCOM !!!",LF,CR,"$"
MSG2 DB "    EE-390,EE-DEPARTMENT,KFUPM,PROGRAM1",LF,CR,"$"
MSG3 DB "PRESS ENTER TO GO OVER THE PROGRAM",LF,CR,"$"
MSG4 DB "PLEASE: ENTER YOUR NAME OR ANY OTHER WORD IN CAPITAL LETTER: ",LF,CR,"$"
MSG5 DB  0DH,0AH,"YOUR ENTERED WORD IN SMALL LETTER IS :",LF,CR
CHAR DB  20 DUP(?),"$"
MSG6 DB "PRESS ENTER TO EXIT TO DOS WINDOW",LF,CR,"$"


.code
MAIN PROC

MOV AX,@DATA
MOV DS,AX

CALL CLEARSCREEN

MOV DH,03
MOV DL,37

CALL SETCURSOR

LEA DX,MSG1
MOV AH,09H
INT 21H

MOV DH,06
MOV DL,20 

CALL SETCURSOR

LEA DX,MSG2
MOV AH,09H
INT 21H

MOV DH,12
MOV DL,25 

CALL SETCURSOR

LEA DX,MSG3
MOV AH,09H
INT 21H



MOV AH,01H
INT 21H
CMP AH,0DH

CALL CLEARSCREEN

MOV DH,06
MOV DL,00

CALL SETCURSOR

MOV AH,09H
LEA SI,MSG4
MOV DX,SI
INT 21H
 
CALL UPPERANDLOWERCASE

MOV DH,08
MOV DL,00

CALL SETCURSOR

MOV AH,09H
LEA SI,MSG5
MOV DX,SI
INT 21H

MOV DH,20
MOV DL,25 

CALL SETCURSOR

LEA DX,MSG6
MOV AH,09H
INT 21H



MOV AH,01H
INT 21H
CMP AH,0DH

CALL CLEARSCREEN


MOV AX, 4C00H
int 21h

MAIN ENDP


CLEARSCREEN PROC

MOV AH,00
MOV AL,03
INT 10H
RET
CLEARSCREEN ENDP


SETCURSOR PROC


MOV AH,02
MOV AL,00
INT 10H
RET
SETCURSOR ENDP


UPPERANDLOWERCASE PROC
    LEA SI,CHAR
LL: MOV AH,01H
    INT 21H
    MOV AH,AL
    ADD AL,20H
    MOV [SI],AL
    INC SI
    CMP AH,0DH
    JNE LL
    RET

UPPERANDLOWERCASE ENDP

END MAIN
