TITLE"prog3"
.MODEL SMALL
.STACK 100
.DATA

        LF EQU 10
        CR EQU 13

        MSG1 DB "WELCOM !!!",LF,CR,"$"
        MSG2 DB "EE-390,EE-DEPARTMENT,KFUPM,PROJECT,082,PROGRAM 3",LF,CR,"$"
	MSG3 DB "PRESS ENTER TO GO OVER THE PROGRAM",LF,CR,"$"
	MSG4 DB "PLEASE: Enter A HEX  VALUE: ",LF,CR,"$"
	MSG5 DB  0DH,0AH,"THE NUMBER OF ONES IS:",LF,CR,"$"
	MSG6 DB "PRESS ENTER TO EXIT TO DOS WINDOW",LF,CR,"$"

        

.CODE
        MAIN PROC
	mov ax,@data
	mov ds,ax


	CALL CLEARSCREEN

	MOV DH,03
	MOV DL,37

	CALL SETCURSOR

	LEA DX,MSG1
	MOV AH,09H
	INT 21H

	MOV DH,06
	MOV DL,20 

	CALL SETCURSOR

	LEA DX,MSG2
	MOV AH,09H
	INT 21H

	MOV DH,12
	MOV DL,25 

	CALL SETCURSOR

	LEA DX,MSG3
	MOV AH,09H
	INT 21H



	MOV AH,01H
	INT 21H
	CMP AH,0DH

	CALL CLEARSCREEN

	MOV DH,06
	MOV DL,00

	CALL SETCURSOR

	MOV AH,09H
	LEA SI,MSG4
	MOV DX,SI
	INT 21H

	CALL HEXNUMI
	

	MOV DH,20
	MOV DL,25 

	CALL SETCURSOR
	
	MOV AH,09H
	LEA DX,MSG6
	MOV AH,09H
	INT 21H
	
	



	MOV AH,01H
	INT 21H
	CMP AH,0DH

	CALL CLEARSCREEN


	MOV AX,4C00H
	INT 21H

        MAIN ENDP


	CLEARSCREEN PROC

	MOV AH,00
	MOV AL,03
	INT 10H
	RET
	CLEARSCREEN ENDP


	SETCURSOR PROC
	
	MOV AH,02
	MOV AL,00
	INT 10H
	RET
	SETCURSOR ENDP
               
	HEXNUMI PROC

	MOV CX,0FFFFH
	C:LOOP C

	MOV AH,01
	INT 21H
	cmp al,39h
	ja A
        sub al,30h
	jmp B
	
	A:sub al,37h
		
	B:mov cx,4
	
	AND AX,0FH	
	mov si,0
	D:rcr AX,1
	jnc E
	inc si
	E:loop D
	MOV DX,SI
	ADD DX,30H
	MOV BX,DX

	MOV DH,08
	MOV DL,00

	CALL SETCURSOR

	MOV AH,9 
	LEA SI,MSG5 
	MOV DX,si 
	INT 21H

	MOV DX,BX	
	EEE:MOV AH,2
	INT 21H
	 
	
	RET
	HEXNUMI ENDP






	END MAIN
