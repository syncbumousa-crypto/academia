TITLE " prog 2"
.MODEL SMALL
.STACK 100 
.DATA


	LF EQU 10
        CR EQU 13

        MSG1 DB "WELCOM !!!",LF,CR,"$"
        MSG2 DB "EE-390,EE-DEPARTMENT,KFUPM,PROJECT,082,PROGRAM 2",LF,CR,"$"
	MSG3 DB "PRESS ENTER TO GO OVER THE PROGRAM",LF,CR,"$"
        MSG4 DB "Enter YOUR NAME THE SET NAME IS (WELCOM):",LF,CR,"$"
	VAR2 DB "WELCOM"	
        MSG5 DB 0DH,0AH,"                                ACCESS GRANTED ",LF,CR,"$"
	MSG6 DB 0DH,0AH,"                               ACCESS NOT GRANTED ",LF,CR,"$"
	VAR1 DB 20 DUP(0),"$"
	MSG7 DB "PRESS ENTER TO EXIT TO DOS WINDOW",LF,CR,"$"





.CODE
MAIN PROC
	ORG 00H
	MOV AX, @DATA
	MOV DS, AX

	CALL CLEARSCREEN

	MOV DH,03
	MOV DL,37

	CALL SETCURSOR

	LEA DX,MSG1
	MOV AH,09H
	INT 21H

	MOV DH,06
	MOV DL,20 

	CALL SETCURSOR

	LEA DX,MSG2
	MOV AH,09H
	INT 21H

	MOV DH,12
	MOV DL,25 

	CALL SETCURSOR

	LEA DX,MSG3
	MOV AH,09H
	INT 21H


	MOV AH,01H
	INT 21H
	CMP AH,0DH



	CALL CLEARSCREEN

	MOV DH,06
	MOV DL,00

	CALL SETCURSOR

	MOV AH,9 
	LEA SI,MSG4 
        MOV DX,SI 
	INT 21H


	LEA DI,VAR1
	LEA SI,VAR2
	CALL INPUT
	LEA DI,VAR1
	LEA SI,VAR2
	CALL CONVERT
	
	MOV DH,20
	MOV DL,25 

	CALL SETCURSOR
	
	MOV AH,09H
	LEA DX,MSG7
	MOV AH,09H
	INT 21H
	
	MOV AH,01H
	INT 21H
	CMP AH,0DH

	CALL CLEARSCREEN


	MOV AX,4C00H
	INT 21H

        MAIN ENDP



CLEARSCREEN PROC

	MOV AH,00
	MOV AL,03
	INT 10H
	RET
	CLEARSCREEN ENDP


	SETCURSOR PROC
	
	MOV AH,02
	MOV AL,00
	INT 10H
	RET
	SETCURSOR ENDP
	
	INPUT PROC NEAR
	A: MOV AH,1
	INT 021H
	MOV [DI],AL
	INC DI
	CMP AL,0DH
	JNZ A

	RET
	INPUT ENDP


	CONVERT PROC NEAR
        B:MOV CX,4H
	cld
	MOV AX,DS
	MOV ES,AX
	MOV AL,[DI]
	MOV DL,[SI]
	rep cmpsb
	jne C

	MOV DH,08
	MOV DL,00

	CALL SETCURSOR

	MOV AH,9 
	LEA SI,MSG5 
	MOV DX,si 
	INT 21H
	JMP E
	C:
	MOV DH,09
	MOV DL,00

	CALL SETCURSOR
	
	MOV AH,9 
	LEA SI,MSG6
	MOV DX,SI
	INT 21H

	E:RET

	CONVERT ENDP


	
END MAIN
