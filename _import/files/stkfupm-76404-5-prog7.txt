TITLE "7"
.MODEL SMALL
.STACK 032H 
.DATA

        MSG1 DB 'Enter A DECIML NUMBER PLEASE: $'	
        MSG2 DB 0DH,0AH,'THE PARITY IS  PO(ODD)  $'
	MSG3 DB 0DH,0AH,'THE PARITY IS  PE(EVEN)  $'
	
.CODE
ORG 00H
MOV AX, @DATA
MOV DS, AX
	MOV AH,9 
	LEA SI,MSG1 
        MOV DX,SI 
	INT 21H

	 MOV AH,1
	INT 021H
	SUB AL,30H
	JP A
	MOV AH,9 
	LEA SI,MSG2 
        MOV DX,SI 
	INT 21H 
	JMP B

	A:MOV AH,9 
	LEA SI,MSG3 
        MOV DX,SI 
	INT 21H 

	B:MOV AH,01
	INT 21H
	
	MOV AH,4CH

	END
	INT 21H
